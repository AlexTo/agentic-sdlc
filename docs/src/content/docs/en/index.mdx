  <Card title="An environment you can trust" icon="approve-check">

  </Card>
  <Card title="Write the spec" icon="pencil">

  </Card>
  <Card title="Agents do the rest" icon="rocket">

  </Card>
  <Card title="You steer" icon="information">

  </Card>
